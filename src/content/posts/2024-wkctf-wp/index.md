找到一些相对全面的 magic 对照表，写脚本对照即可，输出头对应python头两位

```python
# 对应成字典
lines = open(r"C:\Users\HK\Desktop\magic_dict.txt", encoding='utf-8').readlines()
magic_dict = {'0x90c0d0a': '30', '0x20d0d0a': '35', '0xc0d0d0a': '35'} # 这三个值原文章没有，手动查找添加的
for line in lines:
    line = line.strip()
    if 'belong' in line:
        magic = line.split('	')[3]
        version = line.split('	')[4].split(' ')[1].replace('.', '')[:2]
        magic_dict[magic] = version

# 补充
lines = open(r"C:\Users\HK\Desktop\magic.txt", encoding='utf-8').readlines()
out = ''
for line in lines:
    line = line.strip()
    out += magic_dict[line]
print(out)
```

import base64
from pyDes import des
def des_decrypt(cipher_text, password):
    secret_key = password
    des_obj = des(secret_key)
    decrypt_str = des_obj.decrypt(base64.b64decode(cipher_text))
    return decrypt_str

# 字典我用的b神工具里带的，挺小的很方便，rockyou的话相对时间长一些
